# Railway 环境变量配置模板
# 在 Railway Web 服务的 Variables 标签页中添加以下变量

# ============================================
# 必需的环境变量
# ============================================

# Spring 环境配置
SPRING_PROFILES_ACTIVE=production

# 端口（Railway 会自动设置，但可以手动指定）
PORT=8080

# ============================================
# 数据库配置
# ============================================
# 方法 1: 使用完整的 DATABASE_URL（推荐）
# 从数据库服务的 "Connect" 标签页获取连接字符串
# 格式: jdbc:mariadb://host:port/database?useUnicode=true&characterEncoding=utf8&useSSL=true&serverTimezone=UTC
DATABASE_URL=jdbc:mariadb://containers-us-xxx.railway.app:3306/railway?useUnicode=true&characterEncoding=utf8&useSSL=true&serverTimezone=UTC

# 方法 2: 分别设置各个部分（如果方法1不工作）
# 从数据库服务的 Variables 标签页获取这些值
DB_USERNAME=root
DB_PASSWORD=your_database_password
DB_HOST=containers-us-xxx.railway.app
DB_PORT=3306
DB_NAME=railway

# ============================================
# JWT 配置
# ============================================

# JWT 密钥（必须设置，至少32字符）
# 生成方法: openssl rand -base64 32
JWT_SECRET=your_secure_jwt_secret_key_at_least_32_characters_long

# JWT 过期时间（毫秒，默认7天）
JWT_EXPIRATION=604800000

# ============================================
# 可选配置
# ============================================

# 文件上传目录（Railway 使用临时目录）
UPLOAD_DIR=/tmp/uploads

# 日志级别（可选）
# LOGGING_LEVEL_ROOT=INFO
# LOGGING_LEVEL_COM_EXAMPLE_TRAVEL=INFO
